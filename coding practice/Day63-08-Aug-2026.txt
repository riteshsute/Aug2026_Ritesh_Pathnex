# Ansible — Install & Enable Nginx


- name: "Install & Enable Nginx on Pathnex"
  hosts: all
  become: yes

  tasks:
    - name: Install nginx
      yum:
        - name: nginx
          state: present
    
    - name: Enable nginx
      service:
        - name: nginx
          state: restarted
          enabled: yes

# Kubernetes — Deployment with 2 Replicas

apiVersion: apps/v1
kind: Deployment
metadata:
  name: "Pathnex"
spec:
  replicas: 2
  selector:
    matchLabels:
      app: "Pathnex-pods"
  template:
    metadata:
      labels:
        app: "Pathnex-pods"
    spec:
      containers:
        - name: app
          image: Nginx
          ports:
            - containerPort: 80
