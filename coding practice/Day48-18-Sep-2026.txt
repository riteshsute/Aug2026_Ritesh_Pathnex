provider "aws" {
    region = "ap-south-1"
}

resource "aws_vpc" "main" {
    cidr_block = "10.0.0.0/16"

        Name = "pathnex-vpc"
    }

resource "aws_subnet" "public" {
    vpc_id  = aws_vpc.main.id
    cidr_block  = "20.0.2.0/24"
    availability_zone = "ap-south-1a"
    map_public_ip_on_launch = true

    tags = {
        Name = "pathnex-public-subnet"
    }

resource "aws_internet_gateway" "igw" {
    vpc_id   = aws_vpc.main.id

    tags = {
        Name = "pathnex-igw"
    }
}

resource "aws_route_table" "public" {
    vpc_id  =  aws_vpc.main.id

    route {
        cidr_block = "0.0.0.0/0"
        gateway_id = aws_internet_gateway.igw.id
    }

    tags = {
        Name = "pathnex-public-route-table"
    }
}


resource "aws_route_table_association" "public" {
    subnet_id  = aws_subnet.public.id
    route_table_id  = aws_route_table.public.id
}

resource "aws_security_group" "ec2_sg" {
    name  =  "pathnex-ec2-sg"

        from_port  = 22

        Name = "pathnex-ec2-sg"

## 2️⃣ Ansible — Install Nginx & Deploy HTML File

--- 

  -  Name:  Install Nginx & Deploy HTML File
     hosts: all 
     become: yes

     tasks:
        - name: Install Nginx
          yum:
            - name: Nginx
              state: present
        - name: Copy HTML File
          copy:
            src: index.HTML
            dest: /usr/share/nginx/html/index.html
            mode: "0644"
            
          
        

## 3️⃣ Kubernetes — Create Deployment + Service

apiVersion: apps/v1
kind: Deployment
metadata:
   name: pod-v1
spec:
  replicas: 2
  selector:
    matchLabels:
      app: pod-v1
  template:
    metadata:
      labels:
        app: pod-v1
    spec:
      containers:
        - name: Nginx
          image: nginx
          ports:
            - containerPort: 80

---

apiVersion: v1
kind: Service
metadata:
  name: nginx-service
spec:
  selector:
    app: pod-v1
  ports:
    - protocol: TCP
      port: 80
      targetPort: 80
  type: LoadBalancer

# Scheduled Pipelines

## 🔹 Jenkins Pipeline — Scheduled Build
You will learn how to **schedule a Jenkins pipeline using cron**.

pipeline {
    agent any
    environment {
        INSTITUTE_NAME = "Pathnex"
    }
    triggers {
        cron('H 2 * * *') // every day at 2 AM
    }
    stages {
        stage('Checkout') {
            steps {
                git url: 'https://github.com/Pathnex/sample-java-app.git'
            }
        }
        stage('Build') {
            steps {
                sh 'mvn clean package'
            }
        }
    }
}
