### Ansible:
- Create user "pathnex-admin"
- Add SSH key
- Set shell to /bin/bash


--- 

- name: Create User and add shh key
  hosts: all
  become: yes

  tasks:
    - name: Create User 
      user:
        - name: 'Pathnex-admin'
        - 0666
    
    - name: add ssh key
      ansible.buildin.key:
         - name: key
           path: /etc/ssh
        
    - name: add ssh key
      ansible.buildin.module:
         - name: shell 
           path: /bin/bash

### Kubernetes:
- Deployment (2 replicas)
- Expose via ClusterIP

apiVersion: apps/v1
kind: Deployment
metadata:
   name: Replica-pods
spec:
  replicas: 2
  selector:
     matchLabel:
       app: pod
  template:
    metadata:
       label:
         app: pod
    containers:
      - name: nginx
        image: nginx
        ports:
          containerPort: 80

---

apiVersion: v1
kind: service
metadata:
   name: svc-pods
spec:
  type: ClusterIP
  selector: pod
  containerPort:
    port: 80
    target_port: 80
